97inline void* HugePageAllocPortable(size_t size, bool& succeeded)
98{
99 succeeded = false;
100#if defined(_WIN32)
101 void* ptr =
102 VirtualAlloc(nullptr, size, MEM_RESERVE | MEM_COMMIT | MEM_LARGE_PAGES, PAGE_READWRITE);
103 if (ptr)
104 succeeded = true;
105 return ptr;
106#else
107 void* ptr = mmap(nullptr, size, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
108 if (ptr == MAP_FAILED)
109 return nullptr;
110
111 // Best-effort hint, if this fails, we still have valid,
112 // page-aligned (though not huge-page-backed) memory from the
113 // mmap itself, so this isn't treated as a hard failure.
114 madvise(ptr, size, MADV_HUGEPAGE);
115 succeeded = true;
116 return ptr;
117#endif
118}
