175 AccumT TrainStep(const std::vector<T>& x, const std::vector<T>& y, int inferenceSteps)
176 {
177 ResetState();
178 Clamp(x);
179 GetTerminalLayer()->ClampState(y);
180
181 for (int t = 0; t < inferenceSteps; ++t)
182 Step(false);
183 UpdateWeights();
184
185 // Fresh, uncaptured energy readout: sweeps 2+3 only (recompute
186 // predictions/errors from the CURRENT z), deliberately without
187 // sweep 1's UpdateState() -- calling the full Step() again here
188 // would apply an extra, unintended settling step beyond
189 // inferenceSteps, same reasoning as GaussSeidelPCNetwork's own
190 // identical comment.
191 AccumT finalEnergy(0);
192 for (auto& l : layers)
193 l->ComputePrediction();
194 for (auto& l : layers)
195 finalEnergy += l->ComputeError();
196
197 GetTerminalLayer()->UnclampState();
198
199 return finalEnergy;
200 }

230 std::vector<T> Predict(const std::vector<T>& x, int inferenceSteps)
231 {

233 Clamp(x);
234
235 // Seed the hidden states with the forward pass. Without this,
236 // inference starts from z=0 and fails to reach the target.
237 ProjectForward();

244
245 std::vector<T> result(count);
