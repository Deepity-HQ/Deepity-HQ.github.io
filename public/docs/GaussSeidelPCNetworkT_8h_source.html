209 std::vector<T> Predict(const std::vector<T>& x, int inferenceSteps)
210 {
211 ResetState();
212 Clamp(x);
213
214 // Seed the hidden states with the forward pass. Without this,
215 // inference starts from z=0 and fails to reach the target.
216 ProjectForward();
217 for (int t = 0; t < inferenceSteps; ++t)
218 Step();
219
220 GaussSeidelPCLayerT<T, AccumT>* terminal = GetTerminalLayer();
221 const T* beliefs = terminal->GetBeliefs();
222 size_t count = terminal->GetBatchSize() * terminal->GetInputSize();
223
224 std::vector<T> result(count);
225 backend->CopyToHost(result.data(), beliefs, count);
226 return result;
227 }
