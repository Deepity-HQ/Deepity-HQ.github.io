44static inline float ActivationScalar(ActivationType type, float z) noexcept
45{
46 switch (type)
47 {
48 case ActivationType::RELU:
49 return z > 0.0f ? z : 0.0f;
50 case ActivationType::GELU:
51 {
52 float zsq = z * z;
53 float inner = MAGIC_GELU_1 * z + (MAGIC_GELU_2 * MAGIC_GELU_1) * zsq * z;
54 float t = Sleef_tanhf_u10(inner);
55 return 0.5f * z * (1.0f + t);
56 }
57 case ActivationType::SIGMOID:
58 return 1.0f / (1.0f + std::exp(-z));
59 case ActivationType::eSIGMOID:
60 return 0.5f + 0.5f * (z / (1.0f + std::fabs(z)));
61 case ActivationType::TANH:
62 return Sleef_tanhf_u10(z);
63 case ActivationType::HARD_TANH:
64 return std::clamp(z, -1.0f, 1.0f);
65 case ActivationType::LINEAR:
66 case ActivationType::NONE:
67 default:
68 return z;
69 }
70}

79static inline float ActivationDerivativeScalar(ActivationType dType, float z) noexcept
80{
81 switch (dType)
82 {
83 case ActivationType::dRELU:
84 return (z > 0.0f) ? 1.0f : 0.0f;
85 case ActivationType::dGELU:
86 {
87 float zsq = z * z;
88 float inner = MAGIC_GELU_1 * z + (MAGIC_GELU_2 * MAGIC_GELU_1) * zsq * z;
89 float t = Sleef_tanhf_u10(inner);
90 float gprime = MAGIC_GELU_1 + (3.0f * MAGIC_GELU_2 * MAGIC_GELU_1) * zsq;
91 return 0.5f * (1.0f + t) + 0.5f * z * gprime * (1.0f - t * t);
92 }
93 case ActivationType::dSIGMOID:
94 {
95 float sig = 1.0f / (1.0f + std::exp(-z));
96 return sig * (1.0f - sig);
97 }
98 case ActivationType::d_eSIGMOID:
99 {
100 float a = 1.0f + std::fabs(z);
101 return 0.5f / (a * a);
102 }
103 case ActivationType::dTANH:
104 {
105 float t = Sleef_tanhf_u10(z);
106 return 1.0f - t * t;
107 }
108 case ActivationType::dHARD_TANH:
109 return (z > -1.0f && z < 1.0f) ? 1.0f : 0.0f;
110 case ActivationType::dLINEAR:
111 case ActivationType::NONE:

157template <typename T>
158static inline T ActivationScalarT(ActivationType type, T z) noexcept
159{
160 switch (type)
161 {
162 case ActivationType::RELU:
163 return z > T(0) ? z : T(0);
164 case ActivationType::GELU:
165 {
166 constexpr T c1 = T(0.7978845608028654), c2 = T(0.044715);
167 T zsq = z * z;
168 T inner = c1 * z + (c2 * c1) * zsq * z;
169 T t = std::tanh(inner);
170 return T(0.5) * z * (T(1) + t);
171 }
172 case ActivationType::SIGMOID:

174 case ActivationType::eSIGMOID:

176 case ActivationType::TANH:
177 return std::tanh(z);
178 case ActivationType::HARD_TANH:
179 return std::clamp(z, T(-1), T(1));
180 case ActivationType::LINEAR:
181 case ActivationType::NONE:

188template <typename T>
189static inline T ActivationDerivativeScalarT(ActivationType dType, T z) noexcept
190{
191 switch (dType)
192 {
193 case ActivationType::dRELU:
194 return (z > T(0)) ? T(1) : T(0);
195 case ActivationType::dGELU:
196 {
197 constexpr T c1 = T(0.7978845608028654), c2 = T(0.044715);
198 T zsq = z * z;
199 T inner = c1 * z + (c2 * c1) * zsq * z;
200 T t = std::tanh(inner);
201 T gprime = c1 + (T(3) * c2 * c1) * zsq;
202 return T(0.5) * (T(1) + t) + T(0.5) * z * gprime * (T(1) - t * t);
203 }
204 case ActivationType::dSIGMOID:
205 {
206 T sig = T(1) / (T(1) + std::exp(-z));
207 return sig * (T(1) - sig);
208 }
209 case ActivationType::d_eSIGMOID:
210 {
211 T a = T(1) + std::fabs(z);
212 return T(0.5) / (a * a);
213 }
214 case ActivationType::dTANH:
215 {
216 T t = std::tanh(z);
217 return T(1) - t * t;
218 }
219 case ActivationType::dHARD_TANH:
220 return (z > T(-1) && z < T(1)) ? T(1) : T(0);
221 case ActivationType::dLINEAR:
222 case ActivationType::NONE:
223 default:
224 return T(1);
225 }
