225 AccumT TrainStep(const std::vector<T>& x, const std::vector<T>& y, int inferenceSteps = 1,
226 bool computeEnergy = true)
227 {
228 ResetState();
229 Clamp(x);
230 GetTerminalLayer()->ClampState(y);
231
232 bool skipRedundantProjection = useEPC;
233 if (skipRedundantProjection)
234 {
235 for (size_t i = 0; i + 1 < layers.size(); ++i)
236 {
237 if (!layers[i]->IsFlZero())
238 {
239 skipRedundantProjection = false;
240 break;
241 }
242 }
243 }
244
245 auto projectAndClampTerminal = [this, skipRedundantProjection]()
246 {
247 if (skipRedundantProjection)
248 return;
249 ProjectForward();
250 CalculateTerminalError();
251 };
252
253 auto settleStep = [this]()
254 {
255 if (useEPC)
256 EPCStep();
257 else
258 Step(false);
259 if (useIPC)
260 {
261 UpdateWeights();
262 for (auto& l : layers)
263 l->InvalidateMuCache();
264 }
265 };
266
267 auto resyncErrorForClassicSettling = [this]()
268 {
269 if (useEPC)
270 return;
271 for (auto& l : layers)
272 l->CalculateState(false);
273 for (auto& l : layers)
274 l->EnsureMuHoldsDerivative();
275 };
276
277 projectAndClampTerminal();
278 DirectFeedbackUpdate();
279 for (int t = 0; t < inferenceSteps; t++)
280 settleStep();
281 if (!useIPC)
282 {
283 resyncErrorForClassicSettling();
284 UpdateWeights();
285 }
286
287 if (computeEnergy)
288 {
289 lastEnergy = AccumT(0);
290 for (auto& l : layers)
291 lastEnergy += l->CalculateState(true);
292 }
293
294 GetTerminalLayer()->UnclampState();
295 return lastEnergy;
296 }

298 std::vector<T> Predict(const std::vector<T>& x, int inferenceSteps)
299 {
300 ResetState();
301 Clamp(x);
302
303 FullConvPCLayerT<T, AccumT>* terminal = GetTerminalLayer();
304
305 ProjectForward();
306 for (int t = 0; t < inferenceSteps; t++)
307 {
308 if (useEPC)
309 EPCStep();
310 else
311 Step();
312 }
313 if (terminal->GetCrossEntropy())
314 {
315 FullConvPCLayerT<T, AccumT>* below = layers[layers.size() - 2].get();
316 below->ComputeMuOnly();
317 size_t n = below->GetBatchSize() * below->GetOutputSize();
318 backend->Copy(terminal->GetBeliefs(), below->GetMu(), n);
319 }
320
321 const T* beliefs = terminal->GetBeliefs();
322 size_t count = terminal->GetBatchSize() * terminal->GetInputSize();
323
324 std::vector<T> result(count);
325 backend->CopyToHost(result.data(), beliefs, count);
326 return result;

329 void Compile()
330 {
331 if (!layers.empty() &&
332 (size_t)layers.back()->GetInChannels() != (size_t)layers.back()->GetTerminalSize())
333 {
334 throw std::invalid_argument(
335 "FullConvPCNetworkT::Compile(): terminalSize (" +
336 std::to_string(layers.back()->GetTerminalSize()) +
337 ") passed to AddLayer() doesn't match the actual terminal layer's inChannels (" +
338 std::to_string(layers.back()->GetInChannels()) + ").");
339 }
340
341#pragma omp parallel
342 {
343 _MM_SET_FLUSH_ZERO_MODE(_MM_FLUSH_ZERO_ON);
344 _MM_SET_DENORMALS_ZERO_MODE(_MM_DENORMALS_ZERO_ON);
345 }
346 size_t total_elements_needed = 0;
347 for (auto& layer : layers)
348 total_elements_needed += layer->GetRequiredElements();
349
350 if (device == DeviceType::DEVICE_CPU)
351 {
352 cpuArena = std::make_unique<MemoryArenaT<T>>(total_elements_needed, false);
353 for (auto& layer : layers)
354 {
355 layer->BindMemory(*cpuArena);
356 layer->SetTerminalLayer(layers.back().get());
357 }
358 }
359#if defined(DEEPITY_USE_CUDA)
360 else
361 {
362 backend->PrepareForBatchSize(batchSize);
363 gpuArena = std::make_unique<DeviceMemoryArenaT<T, AccumT>>(backend.get(),
364 total_elements_needed);
365 for (auto& layer : layers)
366 {
367 layer->BindMemory(*gpuArena);
368 layer->SetTerminalLayer(layers.back().get());
369 }
370 for (auto& layer : layers)
371 layer->PrepareConvWorkspace();
372 }
373#endif
374
375 if (!useMuPCScaling && !useResidualConnections)
376 return;
377
378 const int L = (int)layers.size() - 1;
379 const int H = L - 1;
380
381 if (L < 1)
382 return;
383
384 if (useMuPCScaling)
385 {
386 const AccumT d_in =
387 AccumT(layers[0]->GetInChannels()) * AccumT(layers[0]->GetKernelH()) *
388 AccumT(layers[0]->GetKernelW());
389
390 AccumT N;
391 AccumT midFanIn;
392 if (H > 0)
393 {
394 N = AccumT(0);
395 midFanIn = AccumT(1);
396 for (int i = 0; i < L - 1; ++i)
397 {
398 AccumT outCh = AccumT(layers[i]->GetOutChannels());
399 if (outCh > N)
400 {
401 N = outCh;
402 midFanIn = outCh * AccumT(layers[i]->GetKernelH()) * AccumT(layers[i]->GetKernelW());
403 }
404 }
405 }
406 else
407 {
408 N = AccumT(layers[L - 1]->GetOutChannels());
409 midFanIn =
410 N * AccumT(layers[L - 1]->GetKernelH()) * AccumT(layers[L - 1]->GetKernelW());
411 }
412
413 for (int l = 1; l <= L; ++l)
414 {
415 AccumT a;
416 if (l == 1)
417 a = std::pow(d_in, AccumT(-0.5));
418 else if (l == L)
419 a = AccumT(1) / N;
420 else
421 a = useResidualConnections ? std::pow(midFanIn * AccumT(L), AccumT(-0.5))
422 : std::pow(midFanIn, AccumT(-0.5));
423
424 layers[l - 1]->SetMuPCScale(a);
425 layers[l - 1]->SetMuPCInit(true);
426 }
427 }
428
429 if (useResidualConnections)
430 {
431 for (int l = 2; l <= H; ++l)
432 {
433 FullConvPCLayerT<T, AccumT>* layer = layers[l - 1].get();
434 bool matches = layer->GetInChannels() == layer->GetOutChannels() &&
435 layer->GetInHeight() == layer->GetOutHeight() &&
436 layer->GetInWidth() == layer->GetOutWidth();
437 if (!matches)
438 {
439 throw std::invalid_argument(
440 "FullConvPCNetworkT::Compile(): residual connections require matching "
441 "channels/spatial dims, layer index " +
442 std::to_string(l - 1) + " has input shape [" +
443 std::to_string(layer->GetInChannels()) + "," + std::to_string(layer->GetInHeight()) +
444 "," + std::to_string(layer->GetInWidth()) + "] but output shape [" +
445 std::to_string(layer->GetOutChannels()) + "," +
446 std::to_string(layer->GetOutHeight()) + "," + std::to_string(layer->GetOutWidth()) +
447 "].");
448 }
449 layer->SetResidual(true);
450 }
451 }
