208 AccumT TrainStep(const std::vector<T>& x, const std::vector<T>& y, int inferenceSteps = 1)
209 {
210 ResetState();
211 Clamp(x);
212 GetTerminalLayer()->ClampState(y);
213
214 ProjectForward();
215 CalculateTerminalError();
216 DirectFeedbackUpdate();
217 for (int t = 0; t < inferenceSteps; t++)
218 Step(false);
219 UpdateWeights();
220
221 AccumT finalEnergy(0);
222 for (auto& l : layers)
223 finalEnergy += l->CalculateState(true);
224
225 GetTerminalLayer()->UnclampState();
226 return finalEnergy;

231 std::vector<T> Predict(const std::vector<T>& x, int inferenceSteps)
232 {
233 ResetState();
234 Clamp(x);

242 size_t count = terminal->GetBatchSize() * terminal->GetInputSize();
243
244 std::vector<T> result(count);

253 void Compile()
254 {
255 if (!layers.empty() && layers.back()->GetInputSize() != layers.back()->GetTerminalSize())
256 {
257 throw std::invalid_argument(
258 "DirectKPPCNetworkT::Compile(): terminalSize (" +
259 std::to_string(layers.back()->GetTerminalSize()) +
260 ") passed to AddLayer() doesn't match the actual terminal layer's size (" +
261 std::to_string(layers.back()->GetInputSize()) + ").");
262 }
263
264 size_t total_elements_needed = 0;
265 for (auto& layer : layers)
266 total_elements_needed += layer->GetRequiredElements();
267
268 backend->PrepareForBatchSize(batchSize);
269
270 if (device == DeviceType::DEVICE_CPU)
271 {
272 cpuArena = std::make_unique<MemoryArenaT<T>>(total_elements_needed, false);
273 for (auto& layer : layers)
274 {
275 layer->BindMemory(*cpuArena);
276 layer->SetTerminalLayer(layers.back().get());
277 }
278 }
279#if defined(DEEPITY_USE_CUDA)
280 else
281 {
282 gpuArena =
283 std::make_unique<DeviceMemoryArenaT<T, AccumT>>(backend.get(), total_elements_needed);
284 for (auto& layer : layers)
285 {
286 layer->BindMemory(*gpuArena);
287 layer->SetTerminalLayer(layers.back().get());
288 }
289 }
290#endif
291 }
