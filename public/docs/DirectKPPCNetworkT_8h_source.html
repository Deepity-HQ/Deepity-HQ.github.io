190 {

193 GetTerminalLayer()->ClampState(y);

196 CalculateTerminalError();

233 void Compile()
234 {
235 if (!layers.empty() && layers.back()->GetInputSize() != layers.back()->GetTerminalSize())
236 {
237 throw std::invalid_argument(
238 "DirectKPPCNetworkT::Compile(): terminalSize (" +
239 std::to_string(layers.back()->GetTerminalSize()) +
240 ") passed to AddLayer() doesn't match the actual terminal layer's size (" +
241 std::to_string(layers.back()->GetInputSize()) + ").");
242 }
243
244 size_t total_elements_needed = 0;
245 for (auto& layer : layers)
246 total_elements_needed += layer->GetRequiredElements();
247
248 backend->PrepareForBatchSize(batchSize);
249
250 if (device == DeviceType::DEVICE_CPU)
251 {
252 cpuArena = std::make_unique<MemoryArenaT<T>>(total_elements_needed, false);
253 for (auto& layer : layers)
254 {
255 layer->BindMemory(*cpuArena);
256 layer->SetTerminalLayer(layers.back().get());
257 }
258 }
259#if defined(DEEPITY_USE_CUDA)
260 else
261 {
262 gpuArena =
263 std::make_unique<DeviceMemoryArenaT<T, AccumT>>(backend.get(), total_elements_needed);
264 for (auto& layer : layers)
265 {
266 layer->BindMemory(*gpuArena);
267 layer->SetTerminalLayer(layers.back().get());
268 }

270#endif
