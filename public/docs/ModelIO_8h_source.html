124 template <typename NetworkType> static bool Load(NetworkType& net, const std::string& filepath)
125 {
126 std::ifstream in(filepath, std::ios::binary);
127 if (!in.is_open())

129

135
136 auto header = nlohmann::json::parse(header_str);
137 size_t base_offset = sizeof(uint64_t) + header_size;
138
139 int layer_idx = 0;
140 for (auto& layerPtr : net.GetLayers())
141 {
142 auto state_dict = layerPtr->GetStateDict();
143 for (auto& [name, tensor] : state_dict)
144 {
145 using ElementType = std::remove_pointer_t<decltype(tensor.data)>;
146 std::string key = "layer_" + std::to_string(layer_idx) + "." + name;
147
148 if (header.contains(key))
149 {
150 std::string fileDtype = header[key].value("dtype", std::string());
151 const char* expectedDtype = DTypeTag<ElementType>::Name();
152 if (!fileDtype.empty() && fileDtype != expectedDtype)
153 {
154 throw std::runtime_error("ModelIO::Load: dtype mismatch for '" + key +
155 "' -- file "
156 "has " +
157 fileDtype + ", network expects " + expectedDtype + ".");
158 }
159
160 size_t start_offset = header[key]["data_offsets"][0];
161 size_t end_offset = header[key]["data_offsets"][1];
162 size_t byte_size = end_offset - start_offset;
163
164 in.seekg(base_offset + start_offset, std::ios::beg);
165 in.read(reinterpret_cast<char*>(tensor.data), byte_size);
166 }
167 }
168 // Pushes the reads above from host staging buffers on to the
169 // actual device-resident parameters. No-op for a CPU layer.
170 detail::SyncToDeviceIfSupported(layerPtr);
171 layer_idx++;
172 }
